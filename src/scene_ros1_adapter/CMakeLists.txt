cmake_minimum_required(VERSION 3.16)
project(xgc_scene_ros1_adapter VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic)

find_package(nlohmann_json 3.7 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(xgc2_math CONFIG REQUIRED)
find_package(Threads REQUIRED)

# The package is a catkin package; the authoring core and the lidar fleet engine need no ROS and also build with plain CMake.
find_package(catkin QUIET)
if(catkin_FOUND)
  catkin_package(INCLUDE_DIRS include LIBRARIES xgc_scene_core xgc_scene_lidar xgc_scene_lidar_ros)
endif()

# Sources and tests are globbed: a new file needs no CMake edit.
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/core/*.cpp)
file(GLOB LIDAR_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/lidar/*.cpp)

if(CORE_SOURCES)
  add_library(xgc_scene_core STATIC ${CORE_SOURCES})
  target_include_directories(xgc_scene_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(xgc_scene_core PUBLIC nlohmann_json::nlohmann_json yaml-cpp OpenSSL::Crypto xgc2_math::scene)
endif()
if(LIDAR_SOURCES)
  add_library(xgc_scene_lidar STATIC ${LIDAR_SOURCES})
  target_include_directories(xgc_scene_lidar PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(xgc_scene_lidar PUBLIC nlohmann_json::nlohmann_json xgc2_math::scene Threads::Threads)
endif()
# Command line tools: src/cli/<name>.cpp is one executable named <name> (finite JSON-in JSON-out tools of the authoring core).
file(GLOB CLI_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/cli/*.cpp)
if(CORE_SOURCES)
  foreach(cli_source ${CLI_SOURCES})
    get_filename_component(cli_name ${cli_source} NAME_WE)
    add_executable(${cli_name} ${cli_source})
    target_link_libraries(${cli_name} PRIVATE xgc_scene_core)
  endforeach()
endif()

# ---- ROS-bound parts: they need catkin and build only inside a catkin workspace; plain CMake builds skip them ----
if(catkin_FOUND)
  # --- service and process (package sa-service) ---
  find_package(XgcXrpc 0.2.0 REQUIRED CONFIG COMPONENTS http)
  find_package(XgcXrpc 0.2.0 REQUIRED CONFIG COMPONENTS json_http)
  file(GLOB SERVICE_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/service/*.cpp)
  add_library(xgc_scene_service STATIC ${SERVICE_SOURCES})
  target_link_libraries(xgc_scene_service PUBLIC xgc_scene_core XgcXrpc::http XgcXrpc::json_http Threads::Threads)

  # Tests below are ordinary gtests. Generated ROS messages need catkin at
  # build time, but neither the service nor the fake outputs need a master.
  include(CTest)
  if(BUILD_TESTING)
    find_package(GTest REQUIRED)
    set(XGC_SCENE_GOLDEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../test/scene_golden" CACHE PATH "golden corpus directory")
    file(GLOB SERVICE_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/test/service/*_test.cpp)
    foreach(test_source ${SERVICE_TESTS})
      get_filename_component(test_name ${test_source} NAME_WE)
      add_executable(${test_name} ${test_source})
      target_link_libraries(${test_name} PRIVATE xgc_scene_service GTest::gtest_main)
      target_compile_definitions(${test_name} PRIVATE XGC_SCENE_GOLDEN_DIR="${XGC_SCENE_GOLDEN_DIR}")
      add_test(NAME ${test_name} COMMAND ${test_name})
    endforeach()
  endif()

  # --- ROS outputs and the simulation client (package sa-ros) ---
  find_package(catkin REQUIRED COMPONENTS roscpp tf2_ros geometry_msgs visualization_msgs std_msgs sensor_msgs
                                         xgc2_geometry_msgs xgc_ros1_adapter_common)
  file(GLOB SIM_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/sim/*.cpp)
  add_library(xgc_scene_sim STATIC ${SIM_SOURCES})
  target_link_libraries(xgc_scene_sim PUBLIC xgc_scene_core XgcXrpc::http XgcXrpc::json_http)
  file(GLOB ROS_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/ros/*.cpp)
  add_library(xgc_scene_ros STATIC ${ROS_SOURCES})
  target_include_directories(xgc_scene_ros PUBLIC ${catkin_INCLUDE_DIRS})
  target_link_libraries(xgc_scene_ros PUBLIC xgc_scene_core xgc_scene_sim ${catkin_LIBRARIES})
  add_dependencies(xgc_scene_ros ${catkin_EXPORTED_TARGETS})
  file(GLOB LIDAR_ROS_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/lidar_ros/*.cpp)
  add_library(xgc_scene_lidar_ros STATIC ${LIDAR_ROS_SOURCES})
  target_link_libraries(xgc_scene_lidar_ros PUBLIC xgc_scene_lidar xgc_scene_ros ${catkin_LIBRARIES})
  add_dependencies(xgc_scene_lidar_ros ${catkin_EXPORTED_TARGETS})
  add_executable(xgc2_scene_adapter src/node_main.cpp)
  target_link_libraries(xgc2_scene_adapter PRIVATE xgc_scene_service xgc_scene_ros xgc_scene_lidar_ros)
  install(TARGETS xgc2_scene_adapter RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})

  if(BUILD_TESTING)
    file(GLOB LIDAR_ROS_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/test/lidar_ros/*_test.cpp)
    foreach(test_source ${LIDAR_ROS_TESTS})
      get_filename_component(test_name ${test_source} NAME_WE)
      add_executable(${test_name} ${test_source})
      target_link_libraries(${test_name} PRIVATE xgc_scene_lidar_ros GTest::gtest_main)
      add_test(NAME ${test_name} COMMAND ${test_name})
    endforeach()
    file(GLOB ROS_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/test/ros/*_test.cpp)
    file(GLOB SIM_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/test/sim/*_test.cpp)
    foreach(test_source ${ROS_TESTS} ${SIM_TESTS})
      get_filename_component(test_name ${test_source} NAME_WE)
      add_executable(${test_name} ${test_source})
      if(test_source IN_LIST ROS_TESTS)
        target_link_libraries(${test_name} PRIVATE xgc_scene_ros xgc_scene_service GTest::gtest_main)
      else()
        target_link_libraries(${test_name} PRIVATE xgc_scene_sim GTest::gtest_main)
      endif()
      add_test(NAME ${test_name} COMMAND ${test_name})
    endforeach()
  endif()

endif()

include(cmake/lidar_control.cmake)

if(catkin_FOUND)
  install(TARGETS xgc_scene_lidar_ros ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
  install(DIRECTORY include/xgc_scene/lidar_ros DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION}/xgc_scene)
  if(TARGET xgc2_scene_adapter)
    install(TARGETS xgc2_scene_adapter
      RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  endif()
  foreach(cli_source ${CLI_SOURCES})
    get_filename_component(cli_name ${cli_source} NAME_WE)
    install(TARGETS ${cli_name}
      RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  endforeach()
  install(FILES process/xgc2-ros1-scene-adapter.json
    DESTINATION "/usr/share/xgc2/process-definitions")
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  # Where the golden corpus of the Python scene service lives (test/scene_golden at the repository root).
  set(XGC_SCENE_GOLDEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../test/scene_golden" CACHE PATH "golden corpus directory")
  file(GLOB CORE_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/test/core/*_test.cpp)
  foreach(test_source ${CORE_TESTS})
    get_filename_component(test_name ${test_source} NAME_WE)
    add_executable(${test_name} ${test_source})
    target_link_libraries(${test_name} PRIVATE xgc_scene_core GTest::gtest_main)
    target_compile_definitions(${test_name} PRIVATE XGC_SCENE_GOLDEN_DIR="${XGC_SCENE_GOLDEN_DIR}")
    add_test(NAME ${test_name} COMMAND ${test_name})
  endforeach()
  file(GLOB LIDAR_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/test/lidar/*_test.cpp)
  foreach(test_source ${LIDAR_TESTS})
    get_filename_component(test_name ${test_source} NAME_WE)
    add_executable(${test_name} ${test_source})
    target_link_libraries(${test_name} PRIVATE xgc_scene_lidar GTest::gtest_main)
    add_test(NAME ${test_name} COMMAND ${test_name})
  endforeach()
endif()
